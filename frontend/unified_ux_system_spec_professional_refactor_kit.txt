## 1. UX SCHEMA (Type Safety & Validation)

### `/ux/schema.ts`

```ts
export type UXRole = "OWNER" | "MEMBER" | "VIEWER"
export type UXPlatform = "desktop" | "mobile"
export type UXSystemID = "people" | "trips" | "notes" | "inbox" | "settings"
export type UXLayoutID = "SYSTEM" | "BOARD" | "FOCUS"
export type UXStateID = "EMPTY" | "LOADING" | "ERROR" | "PARTIAL" | "SUCCESS"

export interface UXSystem {
  id: UXSystemID
  label: string
  icon: string
  subsystems: string[]
}

export interface UXNavSpec {
  type: "sidebar" | "bottom-bar"
  maxPrimary?: number
  systems: UXSystemID[]
}

export interface UXLayoutSpec {
  header: boolean
  sidebar: boolean
  contextPanel: boolean
}

export interface UXComponentSpec {
  slots?: string[]
  types?: string[]
  actions?: string[]
  drag?: boolean
  states?: string[]
}

export interface UXFeatureFlag {
  enabled: boolean
  beta?: boolean
  rollout?: "friends" | "beta" | "public"
}

export interface UXFlow {
  id: string
  steps: string[]
}
```

## 17. UX RUNTIME ENGINE

### `/ux/runtime.ts`

```ts
import { FEATURES } from "./features"
import { PERMISSIONS } from "./permissions"

export function canAccess(userRole: keyof typeof PERMISSIONS, action: string) {
  return PERMISSIONS[userRole]?.includes(action)
}

export function isFeatureEnabled(feature: string) {
  return FEATURES[feature]?.enabled === true
}

export function getFeatureMeta(feature: string) {
  return FEATURES[feature]
}
```

# Integration Example (React / Next.js)

```tsx
import { UX_SYSTEMS, DESKTOP_NAV, isFeatureEnabled } from "@/ux"

export function Sidebar() {
  return (
    <nav>
      {DESKTOP_NAV.systems.map(id => {
        const sys = UX_SYSTEMS[id]
        return (
          <button key={id}>
            <span>{sys.icon}</span>
            <span>{sys.label}</span>
          </button>
        )
      })}

      {isFeatureEnabled("AI_AGENT") && (
        <button>AI Assistant</button>
      )}
    </nav>
  )
}
```

## Refactor Strategy

1. Wrap existing pages with `SystemLayout`
2. Replace hardcoded nav with `DESKTOP_NAV` / `MOBILE_NAV`
3. Centralize feature toggles via `FEATURES`
4. Replace form errors/loading with `VIEW_STATES`
5. Wire telemetry to UX_EVENTS

---


generate a **drop-in adapter layer** so this UX system plugs into the repo with minimal refactor effort.
